# The step critic's READ-ONLY inspection tools (operator directive: the judge gets real tools, not
# just a snapshot). cria executes these deterministically — the judge asks, code reads the disk,
# the result goes back as protocol. Tool descriptions + steers. Parsed by prompts.load_map.
list_dir = List a directory INSIDE the workspace: entries with sizes (dirs marked with /). Call with no path (or ".") for the workspace root. Use this to check what actually exists before ruling on a step that promises a file.
read_file = Read a file INSIDE the workspace and see its REAL content (optionally only lines start_line..end_line, 1-based inclusive). Use this to check that a claimed artifact actually contains what the step requires — a file can exist and still be empty or a stub.
answer_now = Answer NOW with ONLY the JSON verdict: {"done": true|false, "reason": "<short>", "proposed_fix": "<brief fix prose>"}. If you did not manage to read something you needed, SAY THAT in the reason — "could not verify X" — and do not state as fact anything you did not actually read. A guess written as an observation is worse than no answer: it is handed to the coder as work to do.
outside = [refused: {{PATH}} is outside the workspace — the judge inspects only the workspace under {{ROOT}}]
unknown_tool = [refused: {{NAME}} is not one of your tools — you have list_dir and read_file; the coder's tools are listed only so the fix you name is one IT can carry out]
missing = [{{PATH}} does not exist]
unknown = [{{PATH}} could not be read right now — this is NOT evidence that it is absent, and you must not treat it as such. Ask again on your next round, or judge from what you do have.]

# not exist, and built a wrong stale-cache theory on that false fact (rule 5b).
is_directory = [{{PATH}} is a DIRECTORY, not a file — use list_dir to see what is inside it]
empty_file = [{{PATH}} exists and is EMPTY (0 bytes)]
# The CONFIRM judge answers after FIVE rounds of its own tool calls, and a JSON object is the
# shape of the tool-call template it has been repeating — so it emits another one. Measured on
# real captured rounds: gemma4 answered the JSON demand 0/10, and this one-word demand 10/10.
# Prefixing the JSON demand with the critic's "you have no tools now" sentence did NOT help
# (1/10) — the pattern it is continuing is in the transcript, not in the instruction. A bare
# word is the one shape that resembles nothing it can call, which is why answer_now_steer
# (prose or the single word ON_TRACK) has never had this failure either. ternary-bonsai answers
# every shape 6/6, so nothing here costs a model that was already fine.
answer_now_consistent = Reply with EXACTLY ONE WORD on the first line — CONSISTENT if the completion claim holds up against what you read, INCONSISTENT if it does not — then, on a second line, one short sentence saying why.
answer_now_steer = Answer NOW with the short unstick directive spoken to the coder ("You ..."), or EXACTLY the one word ON_TRACK — never JSON, never a verdict.
# The VERDICT tool. A judge was offered list_dir/read_file on the tool channel and required to answer
# on the text channel, with the coder's own tool signatures printed below in the transcript. It did
# the obvious thing and answered where it had been speaking: `<function=satisfied>`,
# `<function=exec_command>`. cria discarded those replies, and "unverified — keep working" reached a
# finished workspace. Answering through a tool is the SAME channel the question was asked in, and the
# answer arrives as a structured tool_call cria reads off the event rather than matching in prose
# (#12). Offered only to judges that declare a verdict key; the steer author, which shares this loop
# and answers in prose, never sees it.
verdict = Give your final answer. Call this once, when you have finished looking. Do not call it and another tool in the same turn.
verdict_satisfied = true only when every requested deliverable is present, verified and working; false otherwise.
verdict_reason = A short evidence-based verdict: what proves completion, or the exact deliverable that is missing, failing, fake or unproven.
verdict_fix = When satisfied is false, NAME the missing or failing deliverable in one line — what must become true, not how to make it true. Never a command, a tool call, a library, a filename the task did not name, or an approach; those belong to the coder. Empty string when true.

rounds_spent = No more inspection rounds are available — {{ROUNDS}} were used, which is the limit here.
inspection_closed = Your last reply asked to use a tool, and there are no inspection rounds left, so nothing was run for it.
