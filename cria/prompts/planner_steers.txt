# Substantive planner steers cria authors for the reasoner (gather phase). Loaded per call.
# The `[note: …]` steers are appended after a leading newline the code adds.
refused_command = [refused: `{{CMD}}` {{WHY}}. During planning the WORKSPACE is read-only (building is the coder's job). To process data you fetched, write it to /tmp instead (e.g. `> /tmp/api.json`) — or read it from the web_fetch result already in the conversation above.]
fresh_note = [note: the workspace `{{CWD}}` does not exist yet — it's a FRESH build with no files to inspect. Plan for the coder to CREATE the project from scratch.]
scratch_note = [note: nothing was saved there yet — a web_fetch returns its content into THIS conversation, not to a file. Read the fetched text above, or save it first with a redirect to /tmp.]
gather_repeat = [you already ran {{TOOL}} with these exact arguments this gather — its result is already above. Don't repeat it: use that result, investigate something DIFFERENT, or if you have enough, output your plan.]
research_first = [before planning, LOOK: read the real thing this work depends on — fetch the source the request names and read what it actually defines, open the files already in the workspace. A plan written without reading is a guess, and whoever builds it follows it literally. Investigate now with your tools; you will be asked for the plan once you have looked. If there is genuinely nothing to read for this request, say so and you will be asked for the plan straight away.]
missing_deliverables = [your plan does not produce everything the request asked for — nothing in it produces: {{MISSING}}. Whoever builds this works through the steps and then stops, so anything no step produces never gets made. Submit the plan again with those covered, keeping the steps you already have.]
host_unread = [your plan builds against {{HOSTS}}, but nothing in this session has read it — so its routes, its parameters and the field names it returns are being assumed, and whoever builds this plan will code those assumptions in and discover them wrong at runtime. Fetch it now and look at what it actually defines, then submit a plan that uses what you found (or, if the plan should read it rather than assume it, say so in the step).]
submit_ungrounded = [your plan names {{URLS}} — but nothing you fetched in this session shows that route exists, so it is a guess. The coder follows the plan verbatim, so a guessed endpoint ships as a bug it then has to discover through a 404. Fetch the real source (the service's own spec/docs) and name only routes it actually defines — or word the step to READ the route from that source instead of naming one. Then submit your plan.]
unknown_tool = [planner has no `{{TOOL}}` tool — you are read-only: exec_command (read), read_file, web_fetch, web_search]
no_search_key = [web_search error: no search API key configured — set BRAVE_SEARCH_API_KEY in the service's env file]
default_ask = Continue and finish the remaining work described in the summary.
