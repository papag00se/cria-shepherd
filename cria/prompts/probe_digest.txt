# proberun's critic-digest fragments (byte-for-byte ports of the Rust upstream). The reasoner/critic
# reads these in the probe digest. Loaded at import — a cria restart re-tunes.
# Format: `<key> = <text>`.  `floor_fmt` keeps a single-brace {x} that Python .format() fills.
floor_fmt = SYNTAX FLOOR: {x}
floor_clean = clean (files parse)
floor_fallback = parse/syntax issues found
floor_none = did not run (no parse checks applicable)
no_probes = PROBES: none ran — no lint/test command was discovered for this project.
exit_clean = exit 0 (ran clean)
exit_no_launch = did NOT launch (tool missing?)
exit_unfinished = UNFINISHED — the harness returned before this check completed; its result is UNKNOWN (treat as not-verified, never as a pass or a failure)
exit_no_tests = no tests collected (nothing to run — not a failure)
# Checks that were selected but whose gate section never came back (script cut short). Absence is a
# GAP, never a pass — without this line a truncated gate read identically to a clean one.
missing_fmt = NOT RUN (no result came back — treat as unverified, not as passing): {x}
